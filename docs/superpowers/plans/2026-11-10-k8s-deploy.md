# k8s 배포 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 수집 서버 프로세스 하나가 다섯 공장의 생성기를 함께 돌리는 통합 실행을 만들고, 접두어·CORS·토큰 설정과 k8s 매니페스트·이미지 빌드 경로를 갖춰 네임스페이스 `u4-2` 에 파드 하나(500m/1Gi)로 올릴 수 있게 한다.

**Architecture:** 수신 핵심을 `PlantContext.receive()` 로 빼서 HTTP 라우트와 새 `LocalSender` 가 함께 쓴다. `--with-simulators` 면 lifespan 이 공장마다 시뮬레이터 태스크를 띄우고, 백필 창을 10분으로 줄여 창마다 이벤트 루프에 양보한다. 배포 설정은 `collector/settings.py` 의 환경 변수, 매니페스트는 `deploy/k8s/` kustomize.

**Tech Stack:** Python 3.11, FastAPI(root_path, CORSMiddleware), kustomize, nginx Ingress, GitHub Actions.

**Spec:** `docs/superpowers/specs/2026-11-10-k8s-deploy-design.md`

## Global Constraints

- 기존 의존성만. 코드 변경은 작게, HTTP 전송 경로는 그대로 남긴다.
- 자원: requests 200m/512Mi, limits 500m/1Gi. 네임스페이스 합계 한도 CPU 2 · 4Gi.
- 보안: `INGEST_TOKEN` 이 있으면 `/ingest` 는 헤더 `X-Ingest-Token` 필수. `LocalSender` 는 토큰을 거치지 않는다.
- 평가용 시나리오(`SIM_SCENARIO_DIR`)는 시뮬레이터에만, 이력 적재에는 절대 주지 않는다(기획서 8장).
- HDS 는 배포에서만 뺀다. 파일·테스트는 남긴다.
- 커밋 메시지는 한국어, 끝에 `Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>`.
- 클러스터 적용·Harbor push 는 하지 않는다(자격 없음). 로컬 검증까지.

"""수신 핵심 PlantContext.receive (k8s 설계 §4.1): HTTP 라우트와 LocalSender 가 함께 쓴다."""
from datetime import datetime, timedelta, timezone

import pytest

from collector.plant_context import PlantContext
from collector.plants import PlantEntry
from common.message import Sample
from common.timefmt import to_db

T0 = datetime(2026, 11, 9, 1, 0, tzinfo=timezone.utc)


@pytest.fixture
def ctx(rfcc_plant_path, rfcc_runtime_path):
    c = PlantContext.open(PlantEntry(key="rfcc", plant=str(rfcc_plant_path), runtime=str(rfcc_runtime_path), db=":memory:"))
    yield c
    c.close()


def _samples(start, n, value=1.45):
    return [Sample(equip_id="P-1101A", tag_id="VI-P-1101A", ts=start + timedelta(seconds=i), value=value, quality="Good") for i in range(n)]


def test_receive_stores_dedupes_and_rejects_old_rows(ctx):
    now = to_db(T0)
    r = ctx.receive("b-1", "T", _samples(T0, 5), now)
    assert (r.stored, r.accepted, r.duplicate) == (5, 5, False)
    assert ctx.receive("b-1", "T", _samples(T0, 5), now).duplicate is True
    ctx.receive("b-2", "T", _samples(T0 + timedelta(days=3), 1), now)          # sim_now 가 3일 앞으로
    r = ctx.receive("b-3", "T", _samples(T0 + timedelta(days=1), 2), now)      # 보관(1일) 밖
    assert r.stored == 0 and r.rejected == 2 and "retention" in r.reasons[0].reason


def test_receive_publishes_only_with_subscribers(ctx, monkeypatch):
    calls = []
    monkeypatch.setattr(ctx.broadcaster, "publish", lambda *a: calls.append(a))
    ctx.receive("b-1", "T", _samples(T0, 3), to_db(T0))
    assert calls == []                                                           # 구독자 없음
    monkeypatch.setattr(ctx.broadcaster, "has_subscribers", lambda: True)
    ctx.receive("b-2", "T", _samples(T0 + timedelta(seconds=10), 3), to_db(T0))
    assert len(calls) == 1 and len(calls[0][1]) == 3
```

"""통합 실행 (k8s 설계 §4.1): 수집 서버 프로세스 안에서 공장 생성기가 돈다."""
import time

import pytest
from fastapi.testclient import TestClient

from collector.app import create_single_app
from collector.settings import Settings


def _wait(client, path, pred, timeout=15.0):
    t = time.monotonic()
    while time.monotonic() - t < timeout:
        r = client.get(path).json()
        if pred(r):
            return r
        time.sleep(0.2)
    raise AssertionError(f"timeout waiting on {path}")


def test_with_simulators_fills_current_values(rfcc_plant_path, rfcc_runtime_path):
    app = create_single_app(rfcc_plant_path, rfcc_runtime_path, ":memory:", settings=Settings(with_simulators=True, time_mode="live"))
    with TestClient(app) as c:
        rows = _wait(c, "/plant/current", lambda r: sum(x["status"] == "Good" for x in r) >= 60)
        assert len(rows) == 81


def test_with_simulators_backfill_keeps_api_responsive(rfcc_plant_path, rfcc_runtime_path, tmp_path):

                            settings=Settings(with_simulators=True, time_mode="backfill", backfill_days=1))
    with TestClient(app) as c:
        slow = 0.0
        for _ in range(20):                       # 백필이 도는 동안 20번 묻는다
            t = time.monotonic(); assert c.get("/plant/health").status_code == 200
            slow = max(slow, time.monotonic() - t); time.sleep(0.3)

"""배포 설정 (k8s 설계 §4.2): ROOT_PATH, CORS_ORIGINS, INGEST_TOKEN."""
from datetime import datetime, timedelta, timezone

from fastapi.testclient import TestClient

from collector.app import create_single_app
from collector.settings import Settings
from common.message import IngestBatch, Sample

T0 = datetime(2026, 11, 9, 1, 0, tzinfo=timezone.utc)


def _batch():
    s = [Sample(equip_id="C-1101", tag_id="VI-C1101", ts=T0 + timedelta(seconds=10 * i), value=2.0, quality="Good") for i in range(3)]
    return IngestBatch(source_id="T", batch_id="b-tok", sent_at=T0, samples=s).model_dump_json()


def test_ingest_token_blocks_http_but_not_local(plant_path, runtime_path):
    settings = Settings(ingest_token="s3cret")
    with TestClient(create_single_app(plant_path, runtime_path, ":memory:", settings=settings)) as c:
        h = {"content-type": "application/json"}
        assert c.post("/plant/ingest", content=_batch(), headers=h).status_code == 403
        assert c.post("/plant/ingest", content=_batch(), headers=h | {"X-Ingest-Token": "nope"}).status_code == 403
        assert c.post("/plant/ingest", content=_batch(), headers=h | {"X-Ingest-Token": "s3cret"}).json()["stored"] == 3
        ctx = c.app.state.plants["plant"]
        assert ctx.receive_local("b-local", "L", [Sample(equip_id="C-1101", tag_id="VI-C1101", ts=T0 + timedelta(minutes=1), value=2.0, quality="Good")]).stored == 1


def test_root_path_prefixes_links_and_redirect(plant_path, runtime_path):
    with TestClient(create_single_app(plant_path, runtime_path, ":memory:", settings=Settings(root_path="/plant-sim")), root_path="/plant-sim") as c:
        assert "href='/plant-sim/plant/'" in c.get("/").text
        assert c.get("/plant", follow_redirects=False).headers["location"] == "/plant-sim/plant/"
        assert c.get("/plant/health").status_code == 200


def test_cors_header_when_origins_set(plant_path, runtime_path):

        assert r.headers.get("access-control-allow-origin") == "https://u4-2.skala-us.com"

"""k8s 매니페스트 (설계 §4.5): 자원 예산과 필수 설정."""
import shutil
import subprocess
from pathlib import Path

import pytest
import yaml

ROOT = Path(__file__).resolve().parent.parent / "deploy" / "k8s"


def _docs():
    return [d for f in ROOT.glob("*.yaml") if f.name not in ("kustomization.yaml", "secret.example.yaml")
            for d in yaml.safe_load_all(f.read_text()) if d]


def test_k8s_manifests_build_and_fit_budget():
    docs = {d["kind"]: d for d in _docs()}
    assert set(docs) >= {"Deployment", "Service", "PersistentVolumeClaim", "Ingress"}
    dep = docs["Deployment"]["spec"]
    assert dep["replicas"] == 1 and dep["strategy"]["type"] == "Recreate"
    c = dep["template"]["spec"]["containers"][0]
    res = c["resources"]
    assert res["requests"] == {"cpu": "200m", "memory": "512Mi"} and res["limits"] == {"cpu": "500m", "memory": "1Gi"}
    assert {p for p in ("startupProbe", "readinessProbe", "livenessProbe")} <= set(c)
    assert c["startupProbe"]["failureThreshold"] * c["startupProbe"]["periodSeconds"] >= 1800
    assert "--with-simulators" in c["args"] and "config/plants.k8s.yaml" in " ".join(c["args"])
    env = {e["name"]: e for e in c["env"]}
    assert env["ROOT_PATH"]["value"] == "/plant-sim" and "secretKeyRef" in env["INGEST_TOKEN"]["valueFrom"]
    assert docs["PersistentVolumeClaim"]["spec"]["storageClassName"] == "ebs-sc"
    ing = docs["Ingress"]
    rule = ing["spec"]["rules"][0]
    assert rule["host"] == "u4-2.skala-us.com" and rule["http"]["paths"][0]["path"].startswith("/plant-sim")
    ann = ing["metadata"]["annotations"]
    assert ann["nginx.ingress.kubernetes.io/proxy-buffering"] == "off" and ann["nginx.ingress.kubernetes.io/rewrite-target"] == "/$2"
    assert "LoadBalancer" not in docs["Service"]["spec"].get("type", "ClusterIP")
    if shutil.which("kubectl"):
        subprocess.run(["kubectl", "kustomize", str(ROOT)], check=True, capture_output=True)
    else:
        pytest.skip("kubectl not installed; kustomize build skipped")
```
